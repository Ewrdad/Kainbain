# Creative Ideas for Refactoring Logging Middleware

## Current State Analysis

The KainBain project currently uses:
- **Fastify's built-in logger** (`fastify({ logger: true })`) in `api/index.js`
- **Console.log** for debugging in frontend (`KainBainContext.jsx`)
- **`app.log.error(err)`** for error handling in the API startup

No centralized logging middleware exists. This is an opportunity to design a comprehensive logging solution.

---

## Idea 1: Structured JSON Logging with Correlation IDs

**Concept**: Every request gets a unique correlation ID that flows through all logs, enabling distributed tracing.

```javascript
// api/middleware/logging.js
import { v4 as uuidv4 } from 'uuid';

export function loggingMiddleware(app) {
  app.addHook('onRequest', async (request, reply) => {
    const correlationId = request.headers['x-correlation-id'] || uuidv4();
    request.correlationId = correlationId;
    reply.header('X-Correlation-ID', correlationId);
    
    request.log = request.log.child({ correlationId });
    request.log.info({ 
      method: request.method, 
      url: request.url,
      ip: request.ip 
    }, 'Request started');
  });

  app.addHook('onResponse', async (request, reply) => {
    request.log.info({ 
      statusCode: reply.statusCode,
      responseTime: reply.elapsedTime 
    }, 'Request completed');
  });

  app.addHook('onError', async (request, reply, error) => {
    request.log.error({ 
      error: error.message,
      stack: error.stack,
      statusCode: reply.statusCode 
    }, 'Request failed');
  });
}
```

**Benefits**: 
- End-to-end request tracing
- Structured JSON output for log aggregation (ELK, Datadog, Loki)
- Easy debugging with correlation IDs in headers

## Idea 2: Multi-Transport Logger with Environment-Aware Output

**Concept**: Different log outputs for different environments (pretty for dev, JSON for prod, silent for tests).

```javascript
// api/config/logger.js
import pino from 'pino';
import pinoPretty from 'pino-pretty';

const isTest = process.env.NODE_ENV === 'test';
const isDevelopment = process.env.NODE_ENV === 'development';

let transport;

if (isTest) {
  transport = { target: 'pino/file', options: { destination: '/dev/null' } };
} else if (isDevelopment) {
  transport = {
    target: 'pino-pretty',
    options: {
      colorize: true,
      translateTime: 'HH:MM:ss Z',
      ignore: 'pid,hostname',
      messageFormat: '{msg} {correlationId}',
    },
  };
} else {
  // Production: JSON to stdout for container log collection
  transport = undefined;
}

export const logger = pino({
  level: process.env.LOG_LEVEL || (isDevelopment ? 'debug' : 'info'),
  transport,
  base: { 
    service: 'kainbain-api',
    version: process.env.npm_package_version,
  },
  redact: {
    paths: ['req.headers.authorization', 'req.headers.cookie', '*.password', '*.token'],
    censor: '[REDACTED]',
  },
});
```

**Usage in Fastify**:
```javascript
// api/index.js
import fastify from 'fastify';
import { logger } from './config/logger.js';

const app = fastify({ logger });
```

**Benefits**: 
- Zero-config environment adaptation
- Automatic PII redaction
- Pretty logs in dev, machine-parseable in prod

## Idea 3: Domain-Specific Log Context Enrichment

**Concept**: Automatically enrich logs with KainBain-specific context (projectId, taskId, sessionId, userId).

```javascript
// api/middleware/contextEnrichment.js
export function contextEnrichmentMiddleware(app) {
  app.addHook('onRequest', async (request, reply) => {
    // Extract context from various sources
    const context = {
      projectId: request.headers['x-project-id'] || 
                 request.query?.projectId || 
                 request.body?.projectId,
      taskId: request.headers['x-task-id'] || 
              request.params?.id || 
              request.body?.taskId,
      sessionId: request.headers['x-session-id'],
      userId: request.headers['x-user-id'], // Future auth integration
    };

    // Filter out undefined values
    Object.keys(context).forEach(key => context[key] === undefined && delete context[key]);

    // Create child logger with context
    request.log = request.log.child(context);
    
    // Also attach to request for handlers to use
    request.kainbainContext = context;
  });
}
```

**Usage in route handlers**:
```javascript
app.post('/tasks', async (request, reply) => {
  // request.log already has projectId, taskId context
  request.log.info({ goal: request.body.goal }, 'Task submitted');
  
  // Access context directly
  const { projectId } = request.kainbainContext;
});
```

**Benefits**: 
- Consistent context across all logs
- Easy filtering in log aggregation (e.g., "show all logs for project X")
- No manual logger.child() calls in handlers

## Idea 4: WebSocket-Aware Logging with Message Flow Tracking

**Concept**: Extend logging to WebSocket connections with message-level granularity.

```javascript
// api/middleware/wsLogging.js
export function wsLoggingMiddleware(app) {
  const wsConnections = new Map();

  app.addHook('onRequest', async (request, reply) => {
    if (request.headers.upgrade === 'websocket') {
      const connectionId = uuidv4();
      request.wsConnectionId = connectionId;
      
      // Create WebSocket-specific logger
      const wsLog = request.log.child({ 
        connectionId,
        transport: 'websocket',
        projectId: request.query.projectId 
      });
      
      wsLog.info({ remoteAddress: request.ip }, 'WebSocket connection initiated');
      wsConnections.set(connectionId, { log: wsLog, connectedAt: Date.now() });
    }
  });

  // Decorate WebSocket with logging
  app.decorate('wsLogMessage', function(socket, direction, message) {
    const connectionId = socket.kainbainConnectionId;
    const conn = wsConnections.get(connectionId);
    if (conn) {
      conn.log.debug({ direction, type: message.type, payloadSize: JSON.stringify(message).length }, 'WS message');
    }
  });

  // Cleanup on close
  app.addHook('onClose', async (instance) => {
    wsConnections.forEach((conn, id) => {
      conn.log.info({ duration: Date.now() - conn.connectedAt }, 'WebSocket connection closed');
    });
    wsConnections.clear();
  });
}
```

**Integration with existing WebSocket handler**:
```javascript
// In api/index.js WebSocket handler
eventBridge.subscribe(id, socket);
socket.kainbainConnectionId = connectionId; // Set from onRequest hook

socket.on('message', (data) => {
  const message = JSON.parse(data);
  app.wsLogMessage(socket, 'inbound', message);
  // ... handle message
});

// When sending
socket.send(JSON.stringify(response));
app.wsLogMessage(socket, 'outbound', response);
```

**Benefits**: 
- Full WebSocket message flow visibility
- Connection lifecycle tracking
- Debugging real-time communication issues

## Idea 5: Structured Audit Logging for Critical Operations

**Concept**: Separate audit log stream for security-sensitive operations (task mutations, project changes, workflow transitions).

```javascript
// api/middleware/auditLogging.js
import pino from 'pino';

// Separate audit logger (can write to different destination)
const auditLogger = pino({
  level: 'info',
  transport: process.env.NODE_ENV === 'development' 
    ? { target: 'pino-pretty', options: { colorize: true, translateTime: 'HH:MM:ss' } }
    : undefined,
  base: { service: 'kainbain-audit' },
});

export function auditLog(event) {
  auditLogger.info({
    ...event,
    timestamp: new Date().toISOString(),
    audit: true,
  }, event.action);
}

// Pre-defined audit events
export const AuditEvents = {
  TASK_CREATED: (context, task) => auditLog({
    action: 'TASK_CREATED',
    projectId: context.projectId,
    taskId: task.id,
    stageId: task.stageId,
    actor: context.userId || 'system',
  }),
  
  TASK_ADVANCED: (context, task, fromStage, toStage) => auditLog({
    action: 'TASK_ADVANCED',
    projectId: context.projectId,
    taskId: task.id,
    fromStage,
    toStage,
    actor: context.userId || 'system',
  }),
  
  PROJECT_CREATED: (context, project) => auditLog({
    action: 'PROJECT_CREATED',
    projectId: project.id,
    projectName: project.name,
    actor: context.userId || 'system',
  }),
  
  WORKFLOW_STARTED: (context, workflowId) => auditLog({
    action: 'WORKFLOW_STARTED',
    projectId: context.projectId,
    workflowId,
    actor: context.userId || 'system',
  }),
  
  SESSION_STARTED: (context, projectId) => auditLog({
    action: 'SESSION_STARTED',
    projectId,
    actor: context.userId || 'system',
  }),
};
```

**Usage in route handlers**:
```javascript
import { AuditEvents } from './middleware/auditLogging.js';

app.post('/tasks', async (request, reply) => {
  const task = taskStore.createTask(projectId, { ... });
  AuditEvents.TASK_CREATED(request.kainbainContext, task);
  // ...
});

app.post('/tasks/:id/advance', async (request, reply) => {
  const task = taskStore.advanceStage(...);
  AuditEvents.TASK_ADVANCED(request.kainbainContext, task, oldStage, newStage);
  // ...
});
```

**Benefits**: 
- Immutable audit trail for compliance
- Separate log stream for security monitoring
- Structured data for SIEM integration

## Idea 6: Performance Logging with Automatic Slow Query Detection

**Concept**: Automatically detect and log slow operations with context.

```javascript
// api/middleware/performanceLogging.js
const SLOW_THRESHOLD_MS = parseInt(process.env.SLOW_LOG_THRESHOLD || '1000', 10);

export function performanceLoggingMiddleware(app) {
  app.addHook('onRequest', async (request, reply) => {
    request.startTime = process.hrtime.bigint();
  });

  app.addHook('onResponse', async (request, reply) => {
    const durationNs = process.hrtime.bigint() - request.startTime;
    const durationMs = Number(durationNs) / 1e6;
    
    const logData = {
      method: request.method,
      url: request.url,
      statusCode: reply.statusCode,
      durationMs: Math.round(durationMs * 100) / 100,
    };

    if (durationMs > SLOW_THRESHOLD_MS) {
      request.log.warn(logData, `Slow request detected (>${SLOW_THRESHOLD_MS}ms)`);
    } else {
      request.log.debug(logData, 'Request performance');
    }

    // Also emit metric for monitoring
    if (app.metrics) {
      app.metrics.histogram('http_request_duration_ms', durationMs, {
        method: request.method,
        route: request.routeOptions?.url || 'unknown',
        statusCode: reply.statusCode,
      });
    }
  });
}

// Decorator for timing specific operations
export function withTiming(logger, operationName, fn) {
  const start = process.hrtime.bigint();
  try {
    return await fn();
  } finally {
    const durationMs = Number(process.hrtime.bigint() - start) / 1e6;
    logger.debug({ operation: operationName, durationMs }, 'Operation timed');
  }
}
```

**Usage**:
```javascript
import { withTiming } from './middleware/performanceLogging.js';

app.post('/tasks', async (request, reply) => {
  const task = await withTiming(request.log, 'createTask', () => 
    taskStore.createTask(projectId, { ... })
  );
  
  await withTiming(request.log, 'promptEngine.curatePrompt', () => 
    promptEngine.curatePrompt(task, stage)
  );
  
  // ...
});
```

**Benefits**: 
- Automatic slow request detection
- Granular operation timing
- Metrics integration ready

## Idea 7: Log Level Control via Admin API

**Concept**: Dynamically adjust log levels at runtime without restart.

```javascript
// api/routes/adminLogging.js
export function adminLoggingRoutes(app) {
  // Get current log levels
  app.get('/admin/logging/levels', async () => {
    return {
      default: app.logger.level,
      loggers: Object.fromEntries(
        Object.entries(app.logger.levels?.values || {}).map(([name, logger]) => [name, logger.level])
      ),
    };
  });

  // Change log level for a specific logger or globally
  app.post('/admin/logging/level', async (request, reply) => {
    const { logger: loggerName, level } = request.body;
    
    const validLevels = ['fatal', 'error', 'warn', 'info', 'debug', 'trace', 'silent'];
    if (!validLevels.includes(level)) {
      return reply.status(400).send({ error: 'Invalid log level' });
    }

    if (loggerName) {
      const logger = app.logger.child({ name: loggerName });
      logger.level = level;
      return { status: 'updated', logger: loggerName, level };
    } else {
      app.logger.level = level;
      return { status: 'updated', level: 'global', level };
    }
  });

  // Get recent logs (for debugging)
  app.get('/admin/logging/recent', async (request, reply) => {
    const { lines = 100, level, correlationId } = request.query;
    // Implementation would depend on log storage (file, Loki, etc.)
    return { message: 'Log retrieval requires log aggregation backend' };
  });
}
```

**Benefits**: 
- Debug production issues without redeploy
- Temporary debug logging for specific components
- Operational flexibility

## Idea 8: Frontend Logging Bridge

**Concept**: Unified logging between frontend and backend with shared correlation IDs.

```javascript
// frontend/src/services/logger.js
const isDevelopment = import.meta.env.DEV;
const API_BASE = import.meta.env.VITE_API_BASE || 'http://localhost:3000';

class FrontendLogger {
  constructor() {
    this.correlationId = null;
    this.sessionId = this.generateSessionId();
    this.logLevel = isDevelopment ? 'debug' : 'warn';
    this.buffer = [];
    this.maxBufferSize = 50;
  }

  generateSessionId() {
    return `frontend-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
  }

  setCorrelationId(id) {
    this.correlationId = id;
  }

  _log(level, message, meta = {}) {
    const entry = {
      timestamp: new Date().toISOString(),
      level,
      service: 'kainbain-frontend',
      sessionId: this.sessionId,
      correlationId: this.correlationId,
      message,
      ...meta,
    };

    // Console output in development
    if (isDevelopment) {
      const style = level === 'error' ? 'color: red' : 
                    level === 'warn' ? 'color: orange' : 'color: gray';
      console.log(`%c[${level.toUpperCase()}]`, style, message, meta);
    }

    // Buffer for batch sending
    this.buffer.push(entry);
    if (this.buffer.length > this.maxBufferSize) {
      this.flush();
    }
  }

  debug(message, meta) { this._log('debug', message, meta); }
  info(message, meta) { this._log('info', message, meta); }
  warn(message, meta) { this._log('warn', message, meta); }
  error(message, meta) { this._log('error', message, meta); }

  async flush() {
    if (this.buffer.length === 0) return;
    
    const logs = this.buffer.splice(0, this.buffer.length);
    try {
      await fetch(`${API_BASE}/logs/client`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ logs }),
      });
    } catch (e) {
      // Silently fail - don't break app if logging fails
      if (isDevelopment) console.warn('Failed to send logs to backend', e);
      // Re-buffer for retry
      this.buffer.unshift(...logs);
    }
  }
}

export const logger = new FrontendLogger();

// Auto-flush on page unload
window.addEventListener('beforeunload', () => logger.flush());

// Periodic flush
setInterval(() => logger.flush(), 30000);
```

**Backend endpoint to receive frontend logs**:
```javascript
// api/routes/clientLogs.js
export function clientLogRoutes(app) {
  app.post('/logs/client', async (request, reply) => {
    const { logs } = request.body;
    if (!Array.isArray(logs)) return reply.status(400).send({ error: 'logs array required' });
    
    // Log with special marker for frontend origin
    logs.forEach(log => {
      app.logger.child({ 
        source: 'frontend',
        sessionId: log.sessionId,
        correlationId: log.correlationId,
      })[log.level](log);
    });
    
    return { status: 'accepted', count: logs.length };
  });
}
```

**Usage in frontend**:
```javascript
// frontend/src/context/KainBainContext.jsx
import { logger } from '../services/logger';

// Set correlation ID from WebSocket connection
ws.onmessage = (event) => {
  const msg = JSON.parse(event.data);
  if (msg.type === 'CONNECTED') {
    logger.setCorrelationId(msg.correlationId);
  }
};

// Replace console.log
logger.debug('WS event received', { type: msg.type, payload: msg.payload });
```

**Benefits**: 
- Unified debugging across frontend/backend
- Correlation IDs span full stack
- Frontend errors visible in backend logs

## Idea 9: Structured Error Logging with Error Codes

**Concept**: Standardized error logging with machine-readable error codes for alerting and dashboards.

```javascript
// api/utils/errorCodes.js
export const ErrorCodes = {
  // Validation errors (4xx)
  VALIDATION_FAILED: 'KB_VALIDATION_FAILED',
  PROJECT_NOT_FOUND: 'KB_PROJECT_NOT_FOUND',
  TASK_NOT_FOUND: 'KB_TASK_NOT_FOUND',
  STAGE_NOT_FOUND: 'KB_STAGE_NOT_FOUND',
  
  // Business logic errors (4xx)
  TASK_ALREADY_COMPLETED: 'KB_TASK_ALREADY_COMPLETED',
  INVALID_STAGE_TRANSITION: 'KB_INVALID_STAGE_TRANSITION',
  SESSION_NOT_RUNNING: 'KB_SESSION_NOT_RUNNING',
  
  // System errors (5xx)
  AGENT_SESSION_FAILED: 'KB_AGENT_SESSION_FAILED',
  MODEL_RUNTIME_ERROR: 'KB_MODEL_RUNTIME_ERROR',
  DATABASE_ERROR: 'KB_DATABASE_ERROR',
  WEBSOCKET_ERROR: 'KB_WEBSOCKET_ERROR',
  
  // External dependency errors
  LLM_API_ERROR: 'KB_LLM_API_ERROR',
  FILE_SYSTEM_ERROR: 'KB_FILE_SYSTEM_ERROR',
};

export class KainBainError extends Error {
  constructor(code, message, details = {}) {
    super(message);
    this.name = 'KainBainError';
    this.code = code;
    this.details = details;
    this.statusCode = code.startsWith('KB_VALIDATION') || code.startsWith('KB_PROJECT') || 
                      code.startsWith('KB_TASK') || code.startsWith('KB_STAGE') ||
                      code.startsWith('KB_TASK_ALREADY') || code.startsWith('KB_INVALID') ||
                      code.startsWith('KB_SESSION') ? 400 : 500;
  }
}

// Error handler middleware
export function errorHandler(app) {
  app.setErrorHandler((error, request, reply) => {
    const isKainBainError = error instanceof KainBainError;
    const code = isKainBainError ? error.code : 'KB_INTERNAL_ERROR';
    const statusCode = isKainBainError ? error.statusCode : 500;
    const details = isKainBainError ? error.details : {};

    // Structured error logging
    request.log.error({
      errorCode: code,
      errorMessage: error.message,
      errorStack: error.stack,
      statusCode,
      details,
      // Context automatically included via contextEnrichmentMiddleware
    }, `Request failed: ${code}`);

    // Client response (don't leak internal details in production)
    const response = {
      error: {
        code,
        message: isKainBainError ? error.message : 'Internal server error',
        ...(process.env.NODE_ENV === 'development' && { stack: error.stack }),
      },
    };

    reply.status(statusCode).send(response);
  });
}
```

**Usage in route handlers**:
```javascript
import { ErrorCodes, KainBainError } from './utils/errorCodes.js';

app.get('/projects/:id', async (request, reply) => {
  const project = projectStore.getProject(request.params.id);
  if (!project) {
    throw new KainBainError(
      ErrorCodes.PROJECT_NOT_FOUND,
      'Project not found',
      { projectId: request.params.id }
    );
  }
  return project;
});
```

**Benefits**: 
- Machine-readable error codes for alerting
- Consistent error response format
- Structured logging for error analysis
- Easy dashboard creation (errors by code)

## Idea 10: Logging Test Utilities

**Concept**: Testing utilities to verify logging behavior in unit/integration tests.

```javascript
// api/tests/loggingTestUtils.js
import { Logger } from 'pino';

export function createTestLogger() {
  const logs = [];
  const logger = new Logger({
    level: 'trace',
    transport: {
      target: 'pino/file',
      options: { destination: '/dev/null' },
    },
  });

  // Override log methods to capture
  const originalMethods = {};
  ['fatal', 'error', 'warn', 'info', 'debug', 'trace'].forEach(level => {
    originalMethods[level] = logger[level].bind(logger);
    logger[level] = (obj, msg) => {
      logs.push({ level, timestamp: new Date().toISOString(), ...obj, message: msg });
      originalMethods[level](obj, msg);
    };
  });

  return {
    logger,
    logs,
    getLogs: (filter = {}) => logs.filter(log => 
      Object.entries(filter).every(([k, v]) => log[k] === v)
    ),
    hasLog: (filter) => logs.some(log => 
      Object.entries(filter).every(([k, v]) => log[k] === v)
    ),
    clear: () => logs.length = 0,
  };
}

// Test helper for correlation IDs
export function withCorrelationId(logger, correlationId, fn) {
  const child = logger.child({ correlationId });
  return fn(child);
}
```
